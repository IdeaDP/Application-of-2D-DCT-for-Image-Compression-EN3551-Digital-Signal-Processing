\subsubsection*{6 Using other K values (like K = 100 or K = 135)? and explanation}
\par Using different K values like 100 or 135 will result in less accurate resolutions in frequency. It is observed that due to the signal length 1792 is not divisible by 100 or 135, the last partition will have less than required number of points. This will result in a less accurate DFT calculation for that partition. This can affect the overall accuracy of the frequency analysis. Therefore, it is recommended to use K values that are factors of the signal length to ensure accurate DFT calculations for all partitions.
\par By zero padding or neglecting the last partition the accurate results can be obtained but there is some loss due to this.

data_set=load('signal155.mat','xn_test');

x = data_set.xn_test;
%Seperating the data set into parts

x_partitions = containers.Map();
x_partitions("S5") = x;

for i = 1:4
    start_index = 1;
    final_index = 2^(i-1)*128;
    x_partitions(sprintf('S%d',i)) = x(start_index:final_index);
    fprintf("KEY %s",sprintf('S%d',i));
    disp(length(x_partitions(sprintf('S%d',i))));
end
figure

fs = 128;

for i =1: length(keys(x_partitions))
    s=x_partitions(sprintf('S%d',i));
    N = length(s);
    figure;
    freq_spectrum = fs*(0:floor(N/2))/N;
    Y = fft(s);
    plot(freq_spectrum,abs(Y(1:floor(N/2)+1)));
    title(sprintf('DFT magnitude plot of first %d partitions',N));
end

%DFT averaging method to remove noise

num_of_partitions = 14;
N = floor(length(x)/num_of_partitions);

fft_sum = zeros(1,N);
for i = 1:num_of_partitions
    start_index = (i-1)*N+1;
    final_index = i*N;

    s = x(start_index:final_index);
    fft_sum = fft_sum + fft(s);
    
end

fft_sum = fft_sum/num_of_partitions;
figure
freq_spectrum = fs*(0:floor(N/2))/N;

plot(freq_spectrum,abs(fft_sum(1:floor(N/2)+1)));
title(sprintf('DFT plot with averaging method (14 partitions)'));

K = 128;

for L = 14:-1:1
    figure;
    X_avg = zeros(1,K);
    for j = 1:L
        seg = x((j-1)*K+1 : j*K);
        X_avg = X_avg + fft(seg);
    end
    X_avg = X_avg / L;
    plot((0:K/2)*(fs/K), abs(X_avg(1:K/2+1)));
    title(sprintf('L = %d', L));
end
\end{lstlisting}
